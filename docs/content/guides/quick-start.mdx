## Run an agent

An agent needs only a model. This one has no tools and no sandbox, so it answers from what the model knows:

<Tabs syncKey="lang">

<TabItem label="Python" icon="seti:python">

```python
import asyncio

from ailoy import Agent


async def main() -> None:
    # For openai, use "openai/gpt-6-luna"
    agent = await Agent.from_spec({"model": "anthropic/claude-haiku-5-5"})

    async for output in agent.run("What is the meaning of hello world?"):
        for part in output["message"]["contents"]:
            if part["type"] == "text":
                print(part["text"])

    await agent.close()


asyncio.run(main())
```

</TabItem>
<TabItem label="Node.js" icon="seti:javascript">

<TabItem label="Rust" icon="seti:rust">

```rust
use ailoy::{
    Agent,
    agent::AgentSpec,
    message::{Message, Part, Role},
};
use futures::StreamExt as _;

#[tokio::main]
async fn main() -> anyhow::Result<()> {
    // For openai, use "openai/gpt-6-luna"
    let spec = AgentSpec::new("anthropic/claude-haiku-5-5");
    let agent = Agent::from_spec(spec, Vec::new(), None).await?;

    let query = Message::new(Role::User).with_contents([Part::text("What is the meaning of hello world?")]);
    let mut stream = std::pin::pin!(agent.run(&query));
    while let Some(output) = stream.next().await {
        let message = output?.message;
        if message.role == Role::Assistant {
            for text in message.contents.iter().filter_map(Part::as_text) {
                println!("{text}");
            }
        }
    }

    agent.close().await?;
    Ok(())
}
```

</TabItem>
</Tabs>

`agent.run` yields each message of the turn as it completes; [Running an agent](/docs/guides/running/) covers what it yields and how to stream it instead.

## Give it a sandbox

The same agent with a sandbox can install software, write and run code, and keep files, in a Linux microVM of its own rather than on your machine. The added lines start virtx's server and give it to Ailoy, which makes the agent's sandbox from a Python image there; then they copy the chart the agent draws out to `population.png` before the agent closes and its sandbox goes with it:

<Tabs syncKey="lang">

<TabItem label="Python" icon="seti:python">

```python ins={2,4,6,10-15,31-32} mark={17-24,26}
import asyncio
from pathlib import Path

import ailoy
from ailoy import Agent
from virtx import Client


async def main() -> None:
    # Starts virtx's server, fetched into its cache the first time.
    client = Client.spawn()
    await client.pull_image("python:3.12-slim-trixie").complete()
    # Each agent's sandbox is made on this server, and removed as the agent closes.
    ailoy.set_virtx(client)

    # For openai, use "openai/gpt-6-luna"
    agent = await Agent.from_spec(
        {
            "model": "anthropic/claude-haiku-5-5",
            "instruction": "Write what you are asked for into /artifacts.",
            "use_sandbox": True,
        },
        sandbox={"create": {"image": {"type": "tag", "tag": "python:3.12-slim-trixie"}}},
    )

    async for output in agent.run("Create a bar chart comparing the populations of European countries and save it to /artifacts/population.png."):
        for part in output["message"]["contents"]:
            if part["type"] == "text":
                print(part["text"])

    sandbox = await client.get_sandbox(agent.sandbox_id)
    Path("population.png").write_bytes(await sandbox.read_file("artifacts/population.png"))
    await agent.close()


asyncio.run(main())
```

</TabItem>

<TabItem label="Rust" icon="seti:rust">

```rust ins={7,11-16,35-37} mark={3,18-22,24}
use ailoy::{
    Agent,
    agent::{AgentSpec, SessionSandbox, default_provider},
    message::{Message, Part, Role},
};
use futures::StreamExt as _;
use virtx::{Client, Files as _, SandboxCreateArgs};

#[tokio::main]
async fn main() -> anyhow::Result<()> {
    // Starts virtx's server, fetched into its cache the first time.
    let client = Client::spawn_virtx([] as [&str; 0]).await?;
    client.pull_image("python:3.12-slim-trixie", None).complete(false).await?;
    // Each agent's sandbox is made on this server, and removed as the agent closes.
    default_provider().write().unwrap().virtx = Some(client.clone());

    // For openai, use "openai/gpt-6-luna"
    let spec = AgentSpec::new("anthropic/claude-haiku-5-5")
        .instruction("Write what you are asked for into /artifacts.")
        .sandbox();
    let sandbox = SessionSandbox::Create(SandboxCreateArgs::new("python:3.12-slim-trixie"));
    let agent = Agent::from_spec(spec, Vec::new(), Some(sandbox)).await?;

    let query = Message::new(Role::User).with_contents([Part::text("Create a bar chart comparing the populations of European countries and save it to /artifacts/population.png.")]);
    let mut stream = std::pin::pin!(agent.run(&query));
    while let Some(output) = stream.next().await {
        let message = output?.message;
        if message.role == Role::Assistant {
            for text in message.contents.iter().filter_map(Part::as_text) {
                println!("{text}");
            }
        }
    }

    let sandbox = client.get_sandbox(agent.sandbox_id().unwrap()).await?;
    let png = sandbox.read_file("artifacts/population.png", 0, None).await?;
    std::fs::write("population.png", png)?;
    agent.close().await?;
    Ok(())
}
```
